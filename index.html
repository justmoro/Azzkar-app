""" تحميل ملفات صوت أذكار المساء من alazkar.today دفعة واحدة، بنفس ترتيب ظهورها في الموقع، عشان تعدّل عليها بـ Audacity براحتك. طريقة الاستخدام: 1. حط الملف ده في أي مجلد على جهازك 2. افتح الطرفية في نفس المجلد ونفّذ: python download_evening_azkar.py 3. هيتنزل كل حاجة في مجلد evening_audio/ بترتيب الأذكار (01، 02، 03...) عشان تعرف تتابعهم بسهولة وأنت بتعدّل """ import urllib.request import os # نفس الترتيب اللي ظهرت بيه الأذكار في الموقع ORDER = [32, 6, 33, 34, 9, 10, 35, 36, 13, 14, 15, 16, 17, 19, 20, 21, 22, 23, 24, 25, 28, 27, 29] def main(): os.makedirs("evening_audio", exist_ok=True) total = len(ORDER) for i, n in enumerate(ORDER, 1): filename = f"{i:02d}_{n}.mp3" # مثال: 01_32.mp3 يعني الذكر الأول، رقمه الأصلي 32 dest = os.path.join("evening_audio", filename) url = f"https://alazkar.today/audio/{n}.mp3" if os.path.exists(dest): print(f"[{i}/{total}] موجود بالفعل، تم التخطي: {filename}") continue try: print(f"[{i}/{total}] تحميل: {filename} ...") req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"}) with urllib.request.urlopen(req, timeout=20) as resp, open(dest, "wb") as f: f.write(resp.read()) except Exception as e: print(f" فشل تحميل {filename}: {e}") print("\nخلصت! افتح مجلد evening_audio/ وابدأ تعدّل بـ Audacity.") print("لما تخلص، ابعتلي الملفات المعدّلة وأنا أحطها في مكانها الصح جوه مجلد audio/ بالمشروع.") if __name__ == "__main__": main()
